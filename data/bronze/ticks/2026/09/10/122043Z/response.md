        now.getUTCMonth(),
        now.getUTCDate(),
        hours,

      WAKE_TIMES[0][1]

        now.getUTCFullYear(),
        now.getUTCMonth(),
        now.getUTCDate(),
        hours,
        minutes

      0,
      0
